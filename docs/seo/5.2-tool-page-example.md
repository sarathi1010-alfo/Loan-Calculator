# AI-Assisted Tool Page Generator

This is an example of what the dynamic `page.tsx` file looks like when fully assembled via our generation logic. It acts as a shell that reads the JSON ontology, configures the component, and injects metadata.

```tsx
// src/app/[cluster]/[slug]/page.tsx
import { notFound } from "next/navigation";
import { EmiCalculator } from "@/components/calculators/EmiCalculator";
import { Breadcrumbs } from "@/components/seo/Breadcrumbs";
import { FAQSection } from "@/components/seo/FAQSection";
import { generateToolMetadata } from "@/lib/seo-metadata";
import fs from "fs";
import path from "path";

// 1. Next.js ISR Settings
export const revalidate = 604800; // 1 week
export const dynamicParams = true;

// 2. Data Fetcher
async function getEntityData(slug: string) {
  const dataPath = path.join(
    process.cwd(),
    "data",
    "generated",
    "seo-pages.json",
  );
  const fileContents = fs.readFileSync(dataPath, "utf8");
  const entities = JSON.parse(fileContents);
  return entities.find((e: any) => e.slug === slug);
}

// 3. Dynamic Metadata
export async function generateMetadata({
  params,
}: {
  params: { slug: string };
}) {
  const data = await getEntityData(params.slug);
  if (!data) return {};
  return generateToolMetadata(data);
}

// 4. The UI Shell
export default async function ToolPage({
  params,
}: {
  params: { slug: string };
}) {
  const entity = await getEntityData(params.slug);
  if (!entity) return notFound();

  return (
    <div className="container max-w-4xl py-10">
      <Breadcrumbs
        clusterName={entity.loanType + " Loan"}
        clusterSlug={entity.loanType + "-loan-calculator"}
        toolName={entity.title}
      />

      <h1 className="text-3xl font-bold mt-6 mb-2">{entity.h1}</h1>
      <p className="text-muted-foreground mb-8">{entity.intro}</p>

      {/* The Configured Deterministic Component */}
      <div className="mb-12">
        <EmiCalculator
          initialAmount={entity.amount}
          initialTenure={entity.tenure}
          initialRate={entity.interestRate}
          hideBankSelector={entity.bank !== "generic"}
        />
      </div>

      <FAQSection faqs={entity.faqs} />
    </div>
  );
}
```
